# Mini-SIEM Environment Configuration
# Copy to .env file and update values for your environment

# ==========================================
# Docker Deployment Notes
# ==========================================
# For Docker deployment, use docker-compose.yml which sets:
# - SIEM_SERVER_PORT=8010
# - Postgres via SIEM_DB_HOST, SIEM_DB_PORT, SIEM_DB_NAME, SIEM_DB_USER, SIEM_DB_PASSWORD
# - Ports exposed: 8010 (API), 8501 (Dashboard), 5514/udp (syslog)
# See docker-compose.yml for container-specific settings

# ==========================================
# API Server Configuration
# ==========================================

# The base URL of the API server (used by agents)
# Change to your actual server IP or domain in production
# For Docker deployment: http://YOUR_HUB_IP:8010
# For local development: http://localhost:8010
SIEM_API_URL=http://localhost:8010

# API authentication key used by agents and by loopback admin tools.
# The dashboard does not put this key in the browser. Humans sign in with a
# passkey (or a one-time login secret) plus an authenticator app.
# Generate with: openssl rand -hex 32
SIEM_API_KEY=default-insecure-key-change-me

# ==========================================
# Operator sign-in (required for the dashboard)
# ==========================================
# Leave SIEM_PUBLIC unset on a private LAN over HTTP.
# Set these before cloudflared publishes the dashboard. The tunnel must
# point only at http://127.0.0.1:8501. Do not publish 8010 or UDP 5514.
# SIEM_PUBLIC=1
# SIEM_ORIGIN=https://heimdall.example.com
# SIEM_RP_ID=heimdall.example.com
# SIEM_TRUST_PROXY=1
# Optional Caddy profile only. A Cloudflare Tunnel does not need SIEM_PUBLIC_HOST.
# SIEM_PUBLIC_HOST=heimdall.example.com
# Optional one-time enrollment token. If unset, the API writes data/bootstrap.token (mode 600).
# SIEM_BOOTSTRAP_TOKEN=

# ==========================================
# Server Configuration
# ==========================================

# FastAPI server host (0.0.0.0 = all interfaces)
SIEM_SERVER_HOST=0.0.0.0

# FastAPI server port
# Changed to 8010 for Docker deployment to avoid port conflicts
SIEM_SERVER_PORT=8010

# Number of uvicorn workers (for production)
SIEM_WORKERS=4

# ==========================================
